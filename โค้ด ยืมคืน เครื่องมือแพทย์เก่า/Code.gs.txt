var SHEET_ID = '1EgyUaj8IS0WUJohS4UwqWhSC-A1e-OyZtWyd15NSN70';
var DRIVE_FOLDER_ID = '1_8dfU0gUu9UHgSdf0t96L2Z_2ciUzB1X'; 

// 🌟 1. รับคำสั่งแบบ GET (ดึงข้อมูลไปแสดงผล)
function doGet(e) {
  try {
    var action = e.parameter.action;
    var result;
    
    if (action === 'getDropdownData') result = getDropdownData();
    else if (action === 'checkHnExists') result = checkHnExists(e.parameter.hn);
    else if (action === 'checkPatientStatusBorrow') result = checkPatientStatusBorrow(e.parameter.patient);
    else if (action === 'getReturnInitData') result = getReturnInitData();
    else if (action === 'getRefDetails') result = getRefDetails(e.parameter.refId);
    else if (action === 'getDashboardData') result = getDashboardData();
    else if (action === 'getAllPatientsForManage') result = getAllPatientsForManage();
    else if (action === 'getPrintData') result = getPrintData(e.parameter.refId);
    else return ContentService.createTextOutput(JSON.stringify({error: 'Invalid API Action'})).setMimeType(ContentService.MimeType.JSON);

    return ContentService.createTextOutput(JSON.stringify(result)).setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({error: err.message})).setMimeType(ContentService.MimeType.JSON);
  }
}

// 🌟 2. รับคำสั่งแบบ POST (บันทึกข้อมูลลง Sheet)
function doPost(e) {
  try {
    var request = JSON.parse(e.postData.contents);
    var action = request.action;
    var data = request.data;
    var result;

    if (action === 'saveTransaction') result = saveTransaction(data);
    else if (action === 'saveNewPatient') result = saveNewPatient(data);
    else if (action === 'updatePatientFullDetails') result = updatePatientFullDetails(request.oldHn, request.newData);
    else return ContentService.createTextOutput(JSON.stringify({status: 'error', message: 'Invalid API Action'})).setMimeType(ContentService.MimeType.JSON);

    return ContentService.createTextOutput(JSON.stringify({status: 'success', message: result})).setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({status: 'error', message: err.message})).setMimeType(ContentService.MimeType.JSON);
  }
}

function checkHnExists(hn) {
  try {
    var ss = SpreadsheetApp.openById(SHEET_ID);
    var sheet = ss.getSheetByName('ลงทะเบียน HN คนไข้');
    if (!sheet || sheet.getLastRow() <= 1) return { exists: false };
    var data = sheet.getRange(2, 1, sheet.getLastRow() - 1, 8).getValues();
    var searchHn = hn.toString().trim();
    for (var i = 0; i < data.length; i++) {
      if (data[i][0].toString().trim() === searchHn) {
        return { exists: true, name: data[i][1], status: data[i][7] || "Admit" };
      }
    }
    return { exists: false };
  } catch (e) { return { exists: false, error: e.message }; }
}

function generateNextRefId() {
  try {
    var ss = SpreadsheetApp.openById(SHEET_ID);
    var sheet = ss.getSheetByName('ยืม-คืน');
    if (!sheet) return "REF-00001";
    var lastRow = sheet.getLastRow();
    if (lastRow <= 1) return "REF-00001";
    var values = sheet.getRange(2, 2, lastRow - 1, 1).getValues().flat();
    var maxNum = 0;
    values.forEach(function(v) {
      if (v && typeof v === 'string' && v.indexOf('REF-') === 0) {
        var n = parseInt(v.replace('REF-', ''), 10);
        if (!isNaN(n) && n > maxNum) maxNum = n;
      }
    });
    return "REF-" + (maxNum + 1).toString().padStart(5, '0');
  } catch (e) { return "REF-00001"; }
}

function saveTransaction(data) {
  var lock = LockService.getScriptLock();
  try {
    lock.waitLock(30000); 
    var ss = SpreadsheetApp.openById(SHEET_ID);
    var logSheet = ss.getSheetByName('ยืม-คืน');
    var equipSheet = ss.getSheetByName('ข้อมูลรายการเครื่องมือแพทย์');
    var pSheet = ss.getSheetByName('ลงทะเบียน HN คนไข้');
    var timestamp = Utilities.formatDate(new Date(), "GMT+7", "dd/MM/yyyy HH:mm:ss");

    var actualRefId = data.refId;
    if (!actualRefId || actualRefId === "" || actualRefId.includes("ระบบจะออกเลข")) {
      actualRefId = generateNextRefId();
    } else {
      actualRefId = actualRefId.toString().trim();
    }
    
    if (data.patient && data.newStatus) {
      var pid = data.patient.split(' - ')[0].trim();
      var pDArr = pSheet.getRange(2, 1, Math.max(1, pSheet.getLastRow()-1), 1).getValues().flat();
      var pIdx = pDArr.map(String).indexOf(pid);
      if (pIdx !== -1) pSheet.getRange(pIdx + 2, 8).setValue(data.newStatus);
    }

    var masterE = equipSheet.getRange(2, 1, Math.max(1, equipSheet.getLastRow()-1), 1).getValues().flat().map(function(s){return s.toString().trim()});
    if (!data.items || data.items.length === 0) {
      logSheet.appendRow([timestamp, actualRefId, "ยืม", data.staff, data.ward, data.patient, "รอดำเนินการเพิ่มรายการ", "ปกติ", "จองไว้", 0, ""]);
      return "จองใบงานสำเร็จ: " + actualRefId;
    }
    data.items.forEach(function(item) {
      var imgUrl = (item.imageRaw && item.imageRaw.length > 100) ? saveFileToDrive(item.imageRaw, actualRefId + "_" + item.type + "_" + item.equipment.replace(/\s+/g, '_') + ".jpg") : "";
      logSheet.appendRow([timestamp, actualRefId, item.type, data.staff, data.ward, data.patient, item.equipment, item.condition || "ปกติ", item.note || "-", Number(item.deposit) || 0, imgUrl]);
      var idx = masterE.indexOf(item.equipment.trim());
      if (idx !== -1) { var r = idx + 2; if (item.type === "ยืม") equipSheet.getRange(r, 2, 1, 3).setValues([["ยืม", data.patient, timestamp]]); else equipSheet.getRange(r, 2, 1, 3).setValues([["ว่าง", "", timestamp]]); }
    });
    return "บันทึกเรียบร้อย: " + actualRefId;
  } catch (e) { throw new Error(e.message); } finally { lock.releaseLock(); }
}

function getDashboardData() {
  try {
    var ss = SpreadsheetApp.openById(SHEET_ID);
    var eSheet = ss.getSheetByName('ข้อมูลรายการเครื่องมือแพทย์');
    var eData = (eSheet && eSheet.getLastRow() > 1) ? eSheet.getRange(2, 1, eSheet.getLastRow()-1, 2).getValues() : [];
    
    var logSheet = ss.getSheetByName('ยืม-คืน');
    var fullList = [];
    var safeRawLogs = [];

    if (logSheet && logSheet.getLastRow() > 1) {
      var logs = logSheet.getDataRange().getValues();
      var rawLogs = logs.slice(1);
      
      safeRawLogs = rawLogs.map(function(row) {
        return row.map(function(cell) {
          if (cell instanceof Date) return cell.getTime();
          return cell;
        });
      });

      var refMap = {};
      
      for (var i = 1; i < logs.length; i++) {
        var id = logs[i][1] ? logs[i][1].toString().trim() : ""; if (!id) continue;
        if (!refMap[id]) {
          var ds = "N/A"; try { ds = Utilities.formatDate(new Date(logs[i][0]), "GMT+7", "dd/MM/yyyy HH:mm"); } catch(e){}
          refMap[id] = { refId: id, patient: logs[i][5], ward: logs[i][4], items: {}, date: ds, rawDate: new Date(logs[i][0]).getTime(), isRes: false, hasAct: false };
        }
        var eqName = logs[i][6] ? logs[i][6].toString().trim() : "";
        var note = logs[i][8] ? logs[i][8].toString().trim() : "-";
        if (eqName === "รอดำเนินการเพิ่มรายการ") { refMap[id].isRes = true; }
        else {
          refMap[id].hasAct = true;
          if (logs[i][2] === "ยืม") {
             var itD = "N/A"; try { itD = Utilities.formatDate(new Date(logs[i][0]), "GMT+7", "dd/MM/yyyy HH:mm"); } catch(e){}
             refMap[id].items[eqName] = { date: itD, note: note }; 
          } else if (logs[i][2] === "คืน") { delete refMap[id].items[eqName.trim()]; }
        }
      }
      for (var k in refMap) {
        var activeArr = []; for (var n in refMap[k].items) activeArr.push({ name: n, date: refMap[k].items[n].date, note: refMap[k].items[n].note });
        var status = (activeArr.length > 0) ? "ค้างยืม" : (refMap[k].hasAct ? "คืนครบแล้ว" : (refMap[k].isRes ? "รอเพิ่มรายการ" : "คืนครบแล้ว"));
        fullList.push({ refId: k, patient: refMap[k].patient, ward: refMap[k].ward, borrowDate: refMap[k].date, items: activeArr, status: status, rawDate: refMap[k].rawDate });
      }
      fullList.sort(function(a, b) { return b.rawDate - a.rawDate; });
    }
    
    var totalECount = eData.length;
    var borrowedCount = eData.filter(function(r){return r[1]==="ยืม"}).length;
    
    return { 
      summary: { total: totalECount, borrowed: borrowedCount, available: totalECount - borrowedCount }, 
      allTransactions: fullList,
      inventory: eData,     
      historyLogs: safeRawLogs  
    };
  } catch (e) { throw new Error(e.message); }
}

function getDropdownData() {
  var ss = SpreadsheetApp.openById(SHEET_ID);
  var result = { patients: [], availableEquips: [], allEquips: [], wards: [], nextRefId: "ระบบจะออกเลขให้อัตโนมัติ" };
  try {
    var pS = ss.getSheetByName('ลงทะเบียน HN คนไข้');
    if(pS && pS.getLastRow() > 1) {
      var pData = pS.getRange(2, 1, pS.getLastRow()-1, 8).getValues();
      pData.sort(function(a, b) { return a[0].toString().localeCompare(b[0].toString(), undefined, {numeric: true}); });
      result.patients = pData.filter(function(r) { return r[7] && r[7].toString().trim() !== "เสียชีวิต"; }).map(function(r) { return r[0] + " - " + r[1]; });
    }
    var eS = ss.getSheetByName('ข้อมูลรายการเครื่องมือแพทย์');
    if(eS && eS.getLastRow() > 1) { 
      var ed = eS.getRange(2, 1, eS.getLastRow()-1, 2).getValues(); 
      result.allEquips = ed.map(function(r){return r[0].toString().trim()}); 
      result.availableEquips = ed.filter(function(r){return r[1]!=="ยืม"}).map(function(r){return r[0].toString().trim()}); 
    }
    var wS = ss.getSheetByName('ข้อมูล Ward ผู้ยืม');
    if(wS && wS.getLastRow() > 1) result.wards = wS.getRange(2, 1, wS.getLastRow()-1, 1).getValues().flat().filter(String);
  } catch(e){}
  return result;
}

function getActiveRefIds() {
  try {
    var ss = SpreadsheetApp.openById(SHEET_ID);
    var logSheet = ss.getSheetByName('ยืม-คืน');
    if (!logSheet || logSheet.getLastRow() <= 1) return [];
    var data = logSheet.getRange(2, 2, logSheet.getLastRow()-1, 6).getValues(); 
    var refMap = {};
    data.forEach(function(row) {
      var id = row[0] ? row[0].toString().trim() : ""; if (!id) return;
      var type = row[1], eq = row[5] ? row[5].toString().trim() : "";
      if (!refMap[id]) refMap[id] = { hasAct: false, isRes: false, bal: {} };
      if (eq === "รอดำเนินการเพิ่มรายการ") { refMap[id].isRes = true; }
      else {
        refMap[id].hasAct = true;
        if (type === "ยืม") refMap[id].bal[eq] = (refMap[id].bal[eq] || 0) + 1;
        else if (type === "คืน") refMap[id].bal[eq] = (refMap[id].bal[eq] || 0) - 1;
      }
    });
    var activeRefs = [];
    for (var id in refMap) {
      var borrowed = false;
      for (var e in refMap[id].bal) { if (refMap[id].bal[e] > 0) borrowed = true; }
      if ((refMap[id].isRes && !refMap[id].hasAct) || borrowed) { activeRefs.push(id); }
    }
    return activeRefs.sort().reverse();
  } catch (e) { return []; }
}

function getRefDetails(refId) {
  try {
    var ss = SpreadsheetApp.openById(SHEET_ID);
    var logSheet = ss.getSheetByName('ยืม-คืน');
    var pSheet = ss.getSheetByName('ลงทะเบียน HN คนไข้');
    var logData = logSheet.getDataRange().getValues();
    var pData = pSheet.getDataRange().getValues();
    var result = { refId: refId, patient: "", ward: "", staff: "", phone: "", address: "", relative: "", relationship: "", status: "", items: [], allItemsForPrint: [] };
    var balanceMap = {}; 
    for (var i = 1; i < logData.length; i++) {
      var rowRef = logData[i][1] ? logData[i][1].toString().trim() : "";
      if (rowRef === refId.toString().trim()) {
        result.staff = logData[i][3]; result.ward = logData[i][4]; result.patient = logData[i][5];
        var eq = logData[i][6] ? logData[i][6].toString().trim() : "";
        if (eq === "รอดำเนินการเพิ่มรายการ" || eq === "") continue;
        if (!balanceMap[eq]) balanceMap[eq] = { name: eq, count: 0, borrowDate: "", returnDate: "", deposit: 0, note: "", photo: "" };
        var type = logData[i][2];
        var dStr = logData[i][0] instanceof Date ? Utilities.formatDate(logData[i][0], "GMT+7", "dd/MM/yyyy HH:mm") : logData[i][0];
        if (type === "ยืม") {
          balanceMap[eq].count += 1; balanceMap[eq].borrowDate = dStr; balanceMap[eq].deposit = Number(logData[i][9]) || 0;
          balanceMap[eq].note = logData[i][8]; balanceMap[eq].photo = convertToDirectLink(logData[i][10]);
        } else if (type === "คืน") { balanceMap[eq].count -= 1; balanceMap[eq].returnDate = dStr; }
      }
    }
    if (result.patient) {
      var pid = result.patient.split(' - ')[0].trim();
      for (var k = 1; k < pData.length; k++) { if (pData[k][0].toString().trim() === pid) { result.relative = pData[k][2]; result.relationship = pData[k][3]; result.address = pData[k][4]; result.phone = pData[k][5]; result.status = pData[k][7]; break; } }
    }
    for (var key in balanceMap) {
      var itm = balanceMap[key];
      result.allItemsForPrint.push({ equipment: itm.name, borrowDate: itm.borrowDate, returnDate: itm.returnDate, status: itm.count > 0 ? "ยังไม่คืน" : "คืนแล้ว", deposit: itm.deposit, note: itm.note });
      if (itm.count > 0) result.items.push({ equipment: itm.name, date: itm.borrowDate, deposit: itm.deposit, note: itm.note, photo: itm.photo });
    }
    return result;
  } catch (e) { throw new Error(e.message); }
}

function getAllPatientsForManage() {
  var ss = SpreadsheetApp.openById(SHEET_ID);
  var sheet = ss.getSheetByName('ลงทะเบียน HN คนไข้');
  if (!sheet || sheet.getLastRow() <= 1) return [];

}

function updatePatientFullDetails(oldHn, newData) {
  var ss = SpreadsheetApp.openById(SHEET_ID);
  var sheet = ss.getSheetByName('ลงทะเบียน HN คนไข้');
  var dArr = sheet.getRange(2, 1, sheet.getLastRow()-1, 1).getValues().flat();
  var idx = dArr.map(String).indexOf(oldHn.toString());
  if (idx === -1) throw new Error("ไม่พบ HN");
  sheet.getRange(idx + 2, 1, 1, 8).setValues([[newData.hn, newData.name, newData.relative, newData.relationship, newData.address, newData.phone, "", newData.status]]);
  return "อัปเดตสำเร็จ";
}

function saveFileToDrive(base64, name) {
  var folder = DriveApp.getFolderById(DRIVE_FOLDER_ID);
  var blob = Utilities.newBlob(Utilities.base64Decode(base64.split(',')[1]), "image/jpeg", name);
  return folder.createFile(blob).setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW).getUrl();
}

function convertToDirectLink(url) {
  if (!url || (url.indexOf('id=') === -1 && url.indexOf('/d/') === -1)) return "";
  var id = (url.indexOf('id=') !== -1) ? url.split('id=')[1].split('&')[0] : url.split('/d/')[1].split('/')[0];
  return "https://drive.google.com/uc?export=view&id=" + id;
}

function checkPatientStatusBorrow(p) { 
  try {
    var result = { hasOutstanding: false, items: [], currentStatus: "Admit" };
    var ss = SpreadsheetApp.openById(SHEET_ID);
    var logS = ss.getSheetByName('ยืม-คืน');
    if(logS && logS.getLastRow() > 1) {
      var logs = logS.getRange(2, 2, logS.getLastRow()-1, 6).getValues();
      var refStatus = {}; 
      logs.forEach(function(r) {
        var rid = r[0] ? r[0].toString().trim() : "";
        var pat = r[4], eq = r[5] ? r[5].toString().trim() : "", type = r[1];
        if(pat === p) {
          if(!refStatus[rid]) refStatus[rid] = { isRes: false, hasReal: false, bal: {} };
          if(eq === "รอดำเนินการเพิ่มรายการ") refStatus[rid].isRes = true;
          else {
            refStatus[rid].hasReal = true;
            if(type === "ยืม") refStatus[rid].bal[eq] = (refStatus[rid].bal[eq] || 0) + 1;
            else if(type === "คืน") refStatus[rid].bal[eq] = (refStatus[rid].bal[eq] || 0) - 1;
          }
        }
      });
      for(var rid in refStatus) {
        var activeCount = 0; for(var e in refStatus[rid].bal) { if(refStatus[rid].bal[e] > 0) activeCount++; }
        if((refStatus[rid].isRes && !refStatus[rid].hasReal) || activeCount > 0) {
          result.hasOutstanding = true; result.items.push(rid + (activeCount > 0 ? " (ค้างส่ง)" : " (ใบจอง)"));
        }
      }
    }
    var pSheet = ss.getSheetByName('ลงทะเบียน HN คนไข้');
    if(pSheet && pSheet.getLastRow() > 1) {
      var pid = p.split(' - ')[0].trim();
      var pData = pSheet.getRange(2, 1, pSheet.getLastRow()-1, 8).getValues();
      var found = pData.find(function(r) { return r[0].toString().trim() === pid; });
      if(found) result.currentStatus = found[7] || "Admit";
    }
    return result;
  } catch(e) { return { hasOutstanding: false, items: [], currentStatus: "Admit" }; }
}
